
import React, { useState } from 'react';
import {StyleSheet, Text, View, TouchableOpacity} from 'react-native';

export default function App() {
  const [copos, setCopos] = useState(0);

  return (
    <View style={styles.container}>

      <Text style={styles.emoji}>🥤</Text>

      <Text style={styles.titulo}>
        Copos de água hoje: {copos}
      </Text>

      <Text style={styles.meta}>
        {copos >= 8
          ? '🎉 Meta diária atingida!'
          : `Faltam ${8 - copos} copos para a meta`}
      </Text>

      <View style={styles.linha}>

        <TouchableOpacity
          style={styles.botao}
          onPress={() => setCopos(valor => valor + 1)}
        >
          <Text style={styles.textoBotao}>
            + Beber 1 Copo
          </Text>
        </TouchableOpacity>

        <TouchableOpacity
          style={styles.botaoZerar}
          onPress={() => setCopos(0)}
        >
          <Text style={styles.textoZerar}>
            Zerar
          </Text>
        </TouchableOpacity>

      </View>

    </View>
  );
}

const styles = StyleSheet.create({

  container: {flex: 1, justifyContent: 'center', alignItems: 'center', backgroundColor: '#fff', padding: 20},

  emoji: {fontSize: 50, marginBottom: 10},

  titulo: {fontSize: 22, fontWeight: 'bold', color: '#0f172a', marginBottom: 10},

  meta: {fontSize: 15, fontWeight: 'bold',color:'#0f172a', marginBottom: 20},

  linha: {flexDirection: 'row', gap: 10},

  botao: {backgroundColor: '#0284c7', paddingVertical: 12, paddingHorizontal: 20, borderRadius: 8},

  textoBotao: {color: '#fff', fontWeight: 'bold', fontSize: 15},

  botaoZerar: {backgroundColor: '#e2e8f0', paddingVertical: 12, paddingHorizontal: 20, borderRadius: 8},

  textoZerar: {color: '#334155', fontWeight: 'bold', fontSize: 15},

});
